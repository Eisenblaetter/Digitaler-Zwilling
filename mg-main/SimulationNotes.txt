    public SimulationMachine(SimConfig config) {

    public SimConfig simulate() {
        initSim();

        // starte vordefinierten ablauf abhängig von config.processName
        // zustandsübergang prüfen ob möglich, dann aktuellen zustand setzen, sleep, dann ggf. aktion
        SimStateTransition[] transitions = MachineXYProcess.getTransitions(processName); // aus HashMap holen

        // main loop, zustandsübergänge mit richtiger zeit ausführen
        for (SimStateTransition transition : transitions) {
            // throw error on not allowed transitions
            checkTransitionAllowed(); // transition to error if not allowed

            int extraTime = getExtraTime(); // individuell für jede maschine, z.b. overheating oder
            richtung/farbe input

            Thread.sleep(transition.baseDuration * config.speedModifier + extraTime); // statt naivem sleep falls
            dazwischen noch aktion könnte man auch interleaving machen von sleep und aktion ausführen
            this.currentState = transition.to;

            // hier lassen sich theoretisch weitere aktionen implementieren wie z.b. attributsänderungen für spezifische
            maschinen
        }

        return finishSim();
    }
